### 2. 批量处理脚本

#### 添加文件头注释脚本（Python）
```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
批量添加文件头注释脚本
用法: python add_file_header.py [目录路径]
"""

import os
import sys
from datetime import datetime

def add_file_header(file_path, module_path):
    """为单个文件添加文件头注释"""
    with open(file_path, 'r', encoding='utf-8', errors='ignore') as f:
        content = f.read()
    
    # 检查是否已有文件头注释
    if '// 文件名:' in content[:500]:
        print(f"跳过（已有注释）: {file_path}")
        return False
    
    # 获取文件名
    file_name = os.path.basename(file_path)
    
    # 生成文件头注释
    header = f"""///////////////////////////////////////////////////////////////////////////////
// 文件名: {file_name}
// 模块: {module_path}
// 功能描述: [TODO: 请填写功能描述]
// 作者: PhotoNest Team
// 创建日期: {datetime.now().strftime('%Y-%m-%d')}
// 修改记录:
//   {datetime.now().strftime('%Y-%m-%d')}  PhotoNest Team  创建文件
///////////////////////////////////////////////////////////////////////////////

"""
    
    # 写入文件
    with open(file_path, 'w', encoding='utf-8') as f:
        f.write(header + content)
    
    print(f"已添加注释: {file_path}")
    return True

def process_directory(directory):
    """递归处理目录下的所有 C++ 源文件"""
    count = 0
    for root, dirs, files in os.walk(directory):
        for file in files:
            if file.endswith(('.cpp', '.h', '.cc')):
                file_path = os.path.join(root, file)
                module_path = os.path.relpath(root, directory)
                if add_file_header(file_path, module_path):
                    count += 1
    return count

if __name__ == '__main__':
    if len(sys.argv) < 2:
        print("用法: python add_file_header.py [目录路径]")
        sys.exit(1)
    
    directory = sys.argv[1]
    count = process_directory(directory)
    print(f"\n总计添加文件头注释: {count} 个文件")
```

### 3. 注释质量检查脚本

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
注释质量检查脚本
用法: python check_comments.py [目录路径]
"""

import os
import re
import sys

def check_file_comments(file_path):
    """检查单个文件的注释质量"""
    issues = []
    
    with open(file_path, 'r', encoding='utf-8', errors='ignore') as f:
        content = f.read()
    
    # 检查文件头注释
    if '// 文件名:' not in content:
        issues.append("缺少文件头注释")
    
    # 检查类注释
    class_matches = re.findall(r'class\s+\w+', content)
    for class_match in class_matches:
        class_name = class_match.split()[-1]
        # 检查是否有类注释
        if f'/// @brief' not in content:
            issues.append(f"类 {class_name} 缺少 @brief 注释")
    
    # 检查公共方法注释
    public_methods = re.findall(r'^\s+(?:virtual\s+)?(?:\w+)\s+(\w+)\s*\(', content, re.MULTILINE)
    for method in public_methods:
        if f'/// @brief' not in content:
            issues.append(f"方法 {method} 可能缺少注释")
    
    return issues

def check_directory(directory):
    """检查目录下所有文件的注释质量"""
    all_issues = {}
    for root, dirs, files in os.walk(directory):
        for file in files:
            if file.endswith(('.cpp', '.h', '.cc')):
                file_path = os.path.join(root, file)
                issues = check_file_comments(file_path)
                if issues:
                    all_issues[file_path] = issues
    return all_issues

if __name__ == '__main__':
    if len(sys.argv) < 2:
        print("用法: python check_comments.py [目录路径]")
        sys.exit(1)
    
    directory = sys.argv[1]
    issues = check_directory(directory)
    
    print(f"注释质量检查报告")
    print(f"=" * 70)
    
    for file_path, file_issues in issues.items():
        print(f"\n文件: {file_path}")
        for issue in file_issues:
            print(f"  - {issue}")
    
    print(f"\n总计发现 {sum(len(v) for v in issues.values())} 个问题")
```
