## 架构最佳实践

### ✅ 单一职责原则 (SRP)

**规则**: 每个类应该只有一个引起变化的原因

```cpp
// ❌ 错误示例: ImageProcessor 做了太多事情

public:
    void LoadImage(const std::string& path);
    void ProcessImage();
    void SaveImage(const std::string& path);
    void UploadToCloud(const std::string& url);  // 不应该在这里
    void SendNotification(const std::string& msg);   // 不应该在这里

// ✅ 正确示例: 分离职责
class ImageLoader {
public:
    std::unique_ptr<ImageData> Load(const std::string& path);

class NotificationService {
public:
    void Send(const std::string& msg);
};
```

```cpp
// ✅ 正确示例: 使用对象池
class ImageDataPool {
public:
    std::unique_ptr<ImageData> Acquire() {
        std::lock_guard<std::mutex> lock(mutex_);
        if (!pool_.empty()) {
            auto obj = std::move(pool_.back());
            pool_.pop_back();
            return obj;
        }
        return std::make_unique<ImageData>();
    }
    
    void Release(std::unique_ptr<ImageData> obj) {
        obj->Reset();  // 重置状态
        std::lock_guard<std::mutex> lock(mutex_);
        if (pool_.size() < kMaxPoolSize) {
            pool_.push_back(std::move(obj));
        }
        // 超过池大小的对象自动释放
    }
    
private:
    std::vector<std::unique_ptr<ImageData>> pool_;
    mutable std::mutex mutex_;
    static constexpr size_t kMaxPoolSize = 100;
};

void ProcessImages(const std::vector<std::string>& image_paths) {
    ImageDataPool pool;
    for (const auto& path : image_paths) {
        auto image = pool.Acquire();  // 从池中获取，无分配
        LoadImage(path, image.get());
        ProcessImage(image.get());
        pool.Release(std::move(image));  // 归还到池
    }
}
```

### ✅ 使用缓存提高性能

```cpp
// ✅ 最佳实践: 实现 LRU 缓存
template<typename Key, typename Value>
class LRUCache {
public:
    explicit LRUCache(size_t max_size) : max_size_(max_size) {}
    
    /// @brief 获取缓存项
    std::shared_ptr<Value> Get(const Key& key) {
        auto it = cache_.find(key);
        if (it == cache_.end()) {
            return nullptr;  // 缓存未命中
        }
        
        // 移动到链表头部（最近使用）
        lru_list_.splice(lru_list_.begin(), lru_list_, it->second.lru_it);
        return it->second.value;
    }
    
    /// @brief 添加缓存项
    void Put(const Key& key, std::shared_ptr<Value> value) {
        auto it = cache_.find(key);
        if (it != cache_.end()) {
            // 更新现有项
            it->second.value = value;
            lru_list_.splice(lru_list_.begin(), lru_list_, it->second.lru_it);
            return;
        }
        
        // 淘汰最久未使用的项
        if (cache_.size() >= max_size_) {
            auto lru_key = lru_list_.back();
            cache_.erase(lru_key);
            lru_list_.pop_back();
        }
        
        // 添加新项
        lru_list_.push_front(key);
        CacheItem item{value, lru_list_.begin()};
        cache_[key] = std::move(item);
    }
    
private:
    struct CacheItem {
        std::shared_ptr<Value> value;
        typename std::list<Key>::iterator lru_it;
    };
    
    size_t max_size_;
    std::unordered_map<Key, CacheItem> cache_;
    std::list<Key> lru_list_;  // 记录使用顺序
};

// 使用示例
LRUCache<std::string, ImageData> cache(1000);  // 缓存 1000 项

auto image = cache.Get("image1.jpg");
if (image == nullptr) {
    image = std::make_shared<ImageData>();
    LoadImage("image1.jpg", image.get());
    cache.Put("image1.jpg", image);
}
```

```cpp
// ❌ 错误示例: 同步加载大图像，阻塞 UI 线程

// ✅ 正确示例: 使用异步加载
class ImageLoader {
public:
    /// @brief 异步加载图像
    void LoadImageAsync(const std::string& path,
                       std::function<void(std::shared_ptr<ImageData>)> callback) {
        // 启动后台线程加载
        std::thread([this, path, callback]() {
            auto image = std::make_shared<ImageData>();
            LoadImage(path, image.get());
            
            {
                std::lock_guard<std::mutex> lock(queue_mutex_);
                task_queue_.push([callback, image]() {
                    callback(image);  // 任务完成后回调
                });
            }
            cv_.notify_one();
        }).detach();
    }
    
    /// @brief 处理完成的任务（在主线程调用）
    void ProcessCompletedTasks() {
        std::queue<std::function<void()>> tasks;
        {
            std::lock_guard<std::mutex> lock(queue_mutex_);
            tasks = std::move(task_queue_);
        }
        
        while (!tasks.empty()) {
            tasks.front()();  // 执行回调
            tasks.pop();
        }
    }
    
private:
    std::queue<std::function<void()>> task_queue_;
    mutable std::mutex queue_mutex_;
    std::condition_variable cv_;
};

// UI 线程中
void OnButtonClick() {
    loader_->LoadImageAsync("large_image.jpg", [this](std::shared_ptr<ImageData> image) {
        // 此回调在后台线程执行，需要抛到 UI 线程
        CefPostTask(TID_UI, base::BindOnce(&MainWindow::DisplayImage, this, image));
    });
}
```

### ✅ 验证输入

```cpp
// ✅ 最佳实践: 验证所有外部输入
class ImageUploader {
public:
    bool UploadImage(const std::string& file_path) {
        // 1. 验证文件路径
        if (file_path.empty()) {
            SetLastError("文件路径为空");
            return false;
        }
        
        // 2. 验证文件扩展名
        std::string ext = GetFileExtension(file_path);
        if (!IsSupportedFormat(ext)) {
            SetLastError("不支持的图像格式: " + ext);
            return false;
        }
        
        // 3. 验证文件大小
        int64_t file_size = GetFileSize(file_path);
        if (file_size > kMaxFileSize) {
            SetLastError("文件过大: " + std::to_string(file_size) + " 字节");
            return false;
        }
        
        // 4. 验证文件内容（检查魔术数字）
        if (!VerifyImageHeader(file_path)) {
            SetLastError("文件内容不是有效的图像文件");
            return false;
        }
        
        // 5. 安全地打开文件
        FILE* file = fopen(file_path.c_str(), "rb");
        if (file == nullptr) {
            SetLastError("无法打开文件: " + file_path);
            return false;
        }
        
        // ... 处理文件
        
        fclose(file);
        return true;
    }
    
private:
    static constexpr int64_t kMaxFileSize = 100 * 1024 * 1024;  // 100MB
    
    bool IsSupportedFormat(const std::string& ext) {
        static const std::vector<std::string> supported = {
            ".jpg", ".jpeg", ".png", ".gif", ".bmp", ".tiff", ".webp"
        };
        return std::find(supported.begin(), supported.end(), 
                         LowerCase(ext)) != supported.end();
    }
    
    bool VerifyImageHeader(const std::string& file_path) {
        // 检查文件头的魔术数字
        FILE* file = fopen(file_path.c_str(), "rb");
        if (file == nullptr) {
            return false;
        }
        
        uint8_t header[8] = {0};
        fread(header, 1, sizeof(header), file);
        fclose(file);
        
        // JPEG: FF D8 FF
        if (header[0] == 0xFF && header[1] == 0xD8 && header[2] == 0xFF) {
            return true;
        }
        
        // PNG: 89 50 4E 47 0D 0A 1A 0A
        if (header[0] == 0x89 && header[1] == 0x50 && header[2] == 0x4E) {
            return true;
        }
        
        // ... 检查其他格式
        
        return false;
    }
};
```

### ✅ 使用日志调试

```cpp
// ✅ 最佳实践: 实现分级日志
enum class LogLevel {
    DEBUG,
    INFO,
    WARNING,
    ERROR
};

class Logger {
public:
    static Logger& GetInstance() {
        static Logger instance;
        return instance;
    }
    
    void SetLevel(LogLevel level) {
        min_level_ = level;
    }
    
    void Log(LogLevel level, const char* file, int line, const std::string& message) {
        if (level < min_level_) {
            return;
        }
        
        std::lock_guard<std::mutex> lock(mutex_);
        
        // 输出到控制台
        std::cout << "[" << GetLevelString(level) << "] "
                  << file << ":" << line << " - "
                  << message << std::endl;
        
        // 输出到文件
        log_file_ << "[" << GetTimestamp() << "] "
                   << "[" << GetLevelString(level) << "] "
                   << file << ":" << line << " - "
                   << message << std::endl;
    }
    
private:
    Logger() : min_level_(LogLevel::INFO) {
        log_file_.open("photolib.log", std::ios::app);
    }
    
    ~Logger() {
        if (log_file_.is_open()) {
            log_file_.close();
        }
    }
    
    std::string GetLevelString(LogLevel level) {
        switch (level) {
            case LogLevel::DEBUG:   return "DEBUG";
            case LogLevel::INFO:    return "INFO";
            case LogLevel::WARNING: return "WARNING";
            case LogLevel::ERROR:   return "ERROR";
            default:                return "UNKNOWN";
        }
    }
    
    std::string GetTimestamp() {
        auto now = std::chrono::system_clock::now();
        auto time_t = std::chrono::system_clock::to_time_t(now);
        char buffer[26];
        ctime_s(buffer, sizeof(buffer), &time_t);
        return buffer;
    }
    
    LogLevel min_level_;
    std::ofstream log_file_;
    std::mutex mutex_;
};

// 日志宏
#define LOG_DEBUG(msg)   Logger::GetInstance().Log(LogLevel::DEBUG,   __FILE__, __LINE__, msg)
#define LOG_INFO(msg)    Logger::GetInstance().Log(LogLevel::INFO,    __FILE__, __LINE__, msg)
#define LOG_WARNING(msg) Logger::GetInstance().Log(LogLevel::WARNING, __FILE__, __LINE__, msg)
```

### ✅ 使用 CefPostTask 跨线程调用

```cpp
// ✅ 最佳实践: 在不同线程中执行任务
class BrowserManager {
public:
    /// @brief 在 UI 线程中执行任务
    void ExecuteOnUIThread(base::OnceCallback<void()> task) {
        if (CefCurrentlyOn(TID_UI)) {
            // 已经在 UI 线程，直接执行
            std::move(task).Run();
        } else {
            // 抛到 UI 线程
            CefPostTask(TID_UI, std::move(task));
        }
    }
    
    /// @brief 在 IO 线程中执行任务
    void ExecuteOnIOThread(base::OnceCallback<void()> task) {
        if (CefCurrentlyOn(TID_IO)) {
            std::move(task).Run();
        } else {
            CefPostTask(TID_IO, std::move(task));
        }
    }
    
    /// @brief 在渲染进程主线程中执行任务
    void ExecuteOnRenderThread(base::OnceCallback<void()> task) {
        if (CefCurrentlyOn(TID_RENDER)) {
            std::move(task).Run();
        } else {
            CefPostTask(TID_RENDER, std::move(task));
        }
    }
};

// 使用示例
void OnImageProcessed(CefRefPtr<CefBrowser> browser, const std::string& result) {
    // 图像处理在后台线程完成，需要抛到 UI 线程更新界面
    CefPostTask(TID_UI, base::BindOnce([](CefRefPtr<CefBrowser> browser, 
                                            const std::string& result) {
        // 在 UI 线程中执行
        CefRefPtr<CefFrame> frame = browser->GetMainFrame();
        frame->ExecuteJavaScript("onImageProcessed('" + result + "');", 
                               CefString(), 0);
    }, browser, result));
}
```
